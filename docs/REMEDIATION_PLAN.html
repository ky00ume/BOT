"""utils/ranks.py — 생활/전투 스킬 랭크 체계 중앙 관리."""
from enum import IntEnum
from typing import Union


class Rank(IntEnum):
    """스킬 랭크. 숫자값으로 대소 비교 가능."""
    PRACTICE = 0   # 연습
    BEGINNER = 1   # 초급
    INTERMEDIATE = 2  # 중급
    ADVANCED = 3   # 상급
    EXPERT = 4     # 전문가
    MASTER = 5     # 달인

    @classmethod
    def from_str(cls, name: str) -> "Rank":
        """한글 랭크명을 Rank enum으로 변환."""
        mapping = {
            "연습": cls.PRACTICE,
            "초급": cls.BEGINNER,
            "중급": cls.INTERMEDIATE,
            "상급": cls.ADVANCED,
            "전문가": cls.EXPERT,
            "달인": cls.MASTER,
        }
        if name not in mapping:
            raise ValueError(f"알 수 없는 랭크: {name}")
        return mapping[name]

    def to_str(self) -> str:
        return ["연습", "초급", "중급", "상급", "전문가", "달인"][self.value]


def rank_gte(current: Union[str, Rank], required: Union[str, Rank]) -> bool:
    """현재 랭크가 요구 랭크 이상인지 확인.

    Args:
        current: 플레이어의 현재 랭크 (한글 또는 Rank)
        required: 요구되는 최소 랭크 (한글 또는 Rank)

    Returns:
        current >= required 이면 True

    Raises:
        ValueError: 유효하지 않은 랭크명이 전달된 경우
    """
    cur = Rank.from_str(current) if isinstance(current, str) else current
    req = Rank.from_str(required) if isinstance(required, str) else required
    return cur >= req

from utils.exceptions import InsufficientResourceError, InventoryFullError


def deduct(self, source: str, gold: int = 0, items: Optional[Dict[str, int]] = None) -> None:
    # 사전 검증: 모든 자원이 충분한지 먼저 확인
    if gold > 0 and self.player.gold < gold:
        raise InsufficientResourceError(
            f"골드 부족: 필요={gold}, 보유={self.player.gold}"
        )
    if items:
        for item_id, count in items.items():
            if self.player.inventory.get(item_id, 0) < count:
                raise InsufficientResourceError(
                    f"아이템 부족: {item_id} 필요={count}, 보유={self.player.inventory.get(item_id, 0)}"
                )

    # 모든 검증 통과 후에만 실제 차감 (원자성 보장)
    try:
        changes: dict = {}
        if gold:
            self.player.gold -= gold
            changes["gold"] = -gold
        if items:
            for item_id, count in items.items():
                self.player.remove_item(item_id, count)
                changes[item_id] = -count
        tx_log.log(self._name(), "TRANSACTION", source, "차감", changes)
    except Exception as e:
        logger.error(f"차감 실패: source={source}, 오류={e}", exc_info=True)
        raise

# storage.py — 수정 후
from utils.exceptions import DatabaseError
from utils.logger import setup_logger

logger = setup_logger('storage')


class StorageEngine:
    def __init__(self, player):
        self.player = player
        self.items = {}
        self.max_capacity = 20
        self._loaded_successfully = False  # 로드 성공 여부 추적
        self._load()

    def _load(self) -> None:
        """DB에서 보관함 데이터 로드.

        Raises:
            DatabaseError: DB 연결 또는 데이터 파싱 실패 시.
        """
        try:
            with get_db_connection() as conn:
                cursor = conn.cursor()
                cursor.execute(
                    "SELECT items, max_capacity FROM storage WHERE user_id = ?",
                    (self.player.user_id,)  # 하드코딩 제거 (3-B 참조)
                )
                row = cursor.fetchone()
                if row:
                    self.items = json.loads(row["items"])
                    self.max_capacity = row["max_capacity"]

    def decorator(render_fn):

                    exc_info=True
                )
                return fallback_fn(*args, **kwargs)
        return wrapper
    return decorator

    self,
    source: str,
    gold: int = 0,
    exp: float = 0.0,
    items: Optional[Dict[str, int]] = None,
) -> None:
    """보상 지급 — 모든 작업이 성공해야만 커밋 (원자성).

    Raises:
        InventoryFullError: 인벤토리가 가득 차서 아이템 추가 실패 시.
    """
    # 1. 사전 검증: 모든 아이템이 수용 가능한지 확인
    if items:
        used, max_slots = self.player.inventory_check()
        new_slots = sum(
            1 for iid in items
            if iid not in self.player.inventory
        )
        if used + new_slots > max_slots:
            raise InventoryFullError(
                f"인벤토리 부족: {used}/{max_slots}, 추가 필요={new_slots}"
            )

    # 2. 스냅샷 저장 (롤백용)
    snapshot = {
        "gold": self.player.gold,
        "exp": getattr(self.player, "exp", 0.0),
        "inventory": dict(self.player.inventory),
    }

    try:
        changes: dict = {}
        if gold:
            self.player.gold += gold
            changes["gold"] = gold
        if exp:
            self.player.exp += exp
            changes["exp"] = exp
        if items:
            for item_id, count in items.items():
                if not self.player.add_item(item_id, count):
                    raise InventoryFullError(f"추가 실패: {item_id}")
                changes[item_id] = count

        tx_log.log(self._name(), "TRANSACTION", source, "보상 지급", changes)

    except Exception as e:
        # 롤백: 스냅샷으로 복원
        self.player.gold = snapshot["gold"]
        self.player.exp = snapshot["exp"]
        self.player.inventory = snapshot["inventory"]

# utils/logger.py — 수정 후
import logging
import os
from logging.handlers import RotatingFileHandler
from pathlib import Path
from typing import Optional

# 프로젝트 루트 기준 상대 경로 (환경변수로 오버라이드 가능)
_DEFAULT_LOG_DIR = Path(__file__).resolve().parent.parent / "logs"
LOG_DIR = Path(os.getenv("BOT_LOG_DIR", str(_DEFAULT_LOG_DIR)))


def setup_logger(name: str, level: Optional[int] = None) -> logging.Logger:
    logger = logging.getLogger(name)
    if logger.handlers:
        return logger

    if level is None:
        debug_mode = os.getenv("DEBUG", "").lower() in ("1", "true", "yes")
        level = logging.DEBUG if debug_mode else logging.INFO
    logger.setLevel(level)

    console_handler = logging.StreamHandler()
    console_handler.setLevel(logging.INFO)
    console_handler.setFormatter(
        logging.Formatter('[%(levelname)s] %(name)s: %(message)s')
    )
    logger.addHandler(console_handler)

    try:
        LOG_DIR.mkdir(parents=True, exist_ok=True)
        file_handler = RotatingFileHandler(
            LOG_DIR / f"{name}.log",
            maxBytes=10 * 1024 * 1024,
            backupCount=5,
            encoding="utf-8",
        )
        file_handler.setLevel(logging.DEBUG)
        file_handler.setFormatter(
            logging.Formatter(
                '[%(asctime)s] %(levelname)s %(name)s:%(lineno)d - %(message)s',
                datefmt='%Y-%m-%d %H:%M:%S',
            )
        )
        logger.addHandler(file_handler)
    except (OSError, PermissionError) as e:
        logger.warning(f"파일 로깅 설정 실패 ({LOG_DIR}): {e}")

    return logger
